3. Add Core Optimization Method
def optimize_coin(self, pair: str, objective_func, n_trials: int = 15):
    """🎯 Optimize parameters for a specific coin"""
    logger.info(f"🚀 [OPTUNA] Starting optimization for {pair} with {n_trials} trials")
    
    try:
        # Clean logging
        optuna.logging.set_verbosity(optuna.logging.WARNING)
        
        # Create study
        study_name = f"{self.strategy_name}_{pair.replace('/', '_').replace(':', '_')}"
        Path("user_data/strategies/optuna_studies").mkdir(parents=True, exist_ok=True)
        
        study = optuna.create_study(
            study_name=study_name,
            direction='maximize',
            sampler=TPESampler(),
            storage=f"sqlite:///user_data/strategies/optuna_studies/optuna_{study_name}.db",
            load_if_exists=True
        )
        
        # Optimize!
        study.optimize(objective_func, n_trials=n_trials)
        
        # Cache results
        self.studies[pair] = study
        self.best_params_cache[pair] = study.best_params
        
        logger.info(f"✅ [OPTUNA] Optimization completed for {pair}")
        logger.info(f"🏆 [OPTUNA] Best value: {study.best_value:.4f}")
        
        return True
    except Exception as e:
        logger.error(f"❌ [OPTUNA] Optimization failed for {pair}: {e}")
        return False

4. Get Optimized Parameters
def get_optimized_params(self, pair: str) -> dict:
    """📊 Get optimized parameters for a pair"""
    if pair in self.best_params_cache:
        return self.best_params_cache[pair]
    
    # Try loading from database
    try:
        study_name = f"{self.strategy_name}_{pair.replace('/', '_').replace(':', '_')}"
        study = optuna.load_study(
            study_name=study_name,
            storage=f"sqlite:///user_data/strategies/optuna_studies/optuna_{study_name}.db"
        )
        self.best_params_cache[pair] = study.best_params
        return study.best_params
    except:
        # Return defaults if no optimization yet
        return {
            'rsi_period': 14,
            'rsi_buy_threshold': 30,
            'rsi_sell_threshold': 70,
            'ma_fast': 10,
            'ma_slow': 25,
            'volume_threshold': 1.5,
        }

6. Auto-Optimization Trigger
def confirm_trade_entry(self, pair: str, order_type: str, amount: float, rate: float, time_in_force: str, current_time, entry_tag, side: str, **kwargs) -> bool:
    # Count trades and trigger optimization
    if pair not in self.trades_since_optimization:
        self.trades_since_optimization[pair] = 0
    
    self.trades_since_optimization[pair] += 1
    
    # 🔄 Auto-optimize every 20 trades
    if self.trades_since_optimization[pair] >= self.optimization_frequency:
        if self.optimization_enabled:
            logger.info(f"🔄 [OPTUNA] Auto-optimization triggered for {pair}")
            # Add your optimization trigger here
            self.trades_since_optimization[pair] = 0
    
    return True
