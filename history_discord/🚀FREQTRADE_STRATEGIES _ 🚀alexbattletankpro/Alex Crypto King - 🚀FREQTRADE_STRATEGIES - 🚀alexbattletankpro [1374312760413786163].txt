# VVRP Integration
        dataframe = calculate_rolling_vvrp(dataframe, lookback_period=120, num_bins=9, max_bars=200)
        dataframe['VVRP_High_Volume_Long'] = 0
        dataframe['VVRP_High_Volume_Short'] = 0
        for i in range(len(dataframe)):
            for j in range(1, 10):
                mid_price = dataframe.get(f'VVRP_Mid_Price_{j}', pd.Series([None]*len(dataframe))).iloc[i]
                buy_bars = dataframe.get(f'VVRP_Buy_Bars_{j}', pd.Series([0]*len(dataframe))).iloc[i]
                sell_bars = dataframe.get(f'VVRP_Sell_Bars_{j}', pd.Series([0]*len(dataframe))).iloc[i]
                price = dataframe['close'].iloc[i]
                atr = dataframe['atr'].iloc[i] if 'atr' in dataframe else 1.0
                if pd.notnull(mid_price) and abs(price - mid_price) < atr * self.tightness_factor.value:
                    if buy_bars - sell_bars > self.dominance_threshold.value:
                        dataframe.at[i, 'VVRP_High_Volume_Long'] = 1
                    if sell_bars - buy_bars > self.dominance_threshold.value:
                        dataframe.at[i, 'VVRP_High_Volume_Short'] = 1
        # === Global filter indicators ===
        dataframe["volatility_range"] = dataframe["high"] - dataframe["low"]
        dataframe["avg_volatility"] = dataframe["volatility_range"].rolling(window=20).mean()
        dataframe["avg_volume"] = dataframe["volume"].rolling(window=20).mean()
        dataframe["uptrend"] = (dataframe["ema50"] > dataframe["ema50"].shift(5)).astype(int)
