def custom_entry_price(self, pair: str, trade: Optional['Trade'], current_time: datetime, proposed_rate: float, entry_tag: Optional[str], side: str, **kwargs) -> float:
    """
    Customizes the entry price for incremental entries.
    If it's a long entry, it increments the price slightly each time to avoid overbuying.
    If it's a short entry, it decrements the price similarly.
    """
    
    dataframe, last_updated = self.dp.get_analyzed_dataframe(pair=pair, timeframe=self.timeframe)
    entry_price = (dataframe['close'].iat[-1] + dataframe['open'].iat[-1] + proposed_rate + proposed_rate) / 4
    logger.info(f"{pair} Using Entry Price: {entry_price} | close: {dataframe['close'].iat[-1]} open: {dataframe['open'].iat[-1]} proposed_rate: {proposed_rate}")

    # Apply different increments for long and short positions
    if entry_tag == 'enter_long':
        # Long position: increment the entry price if we re-enter close to the last entry
        if self.last_entry_price is not None and abs(entry_price - self.last_entry_price) < 0.0001:
            entry_price *= (1 + self.increment.value)  # Increment by the defined percentage
            logger.info(f'{pair} Incremented long entry price to: {entry_price}, previous: {self.last_entry_price}')
    
    elif entry_tag == 'enter_short':
        # Short position: decrement the entry price for incremental short entries
        if self.last_entry_price is not None and abs(entry_price - self.last_entry_price) < 0.0001:
            entry_price *= (1 - self.increment.value)  # Decrement by the defined percentage
            logger.info(f'{pair} Decremented short entry price to: {entry_price}, previous: {self.last_entry_price}')

    # Store the last entry price for future comparisons
    self.last_entry_price = entry_price

    return entry_price
