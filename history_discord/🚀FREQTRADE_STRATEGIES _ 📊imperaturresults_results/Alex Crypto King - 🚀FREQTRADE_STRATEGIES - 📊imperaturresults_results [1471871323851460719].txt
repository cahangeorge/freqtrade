    def custom_stoploss(self, pair: str, trade: Trade, current_time: datetime,
                        current_rate: float, current_profit: float, **kwargs) -> float:
        dataframe, _ = self.dp.get_analyzed_dataframe(pair, self.timeframe)
        if len(dataframe) < 1:
            return self.stoploss

        atr_pct = dataframe.iloc[-1].get('atr_pct', 2.0) / 100
        dynamic_sl = min(-0.04, -atr_pct * 2)

        # Breakeven protection - don't lose money once in profit
        if current_profit > 0.01:  # 1% profit
            return -0.01  # Stop at breakeven + small buffer
        
        # Small profit protection
        if current_profit > 0.03:  # 3% profit
            return 0.01   # Lock in 1% minimum
            
        # Medium profit trailing
        if current_profit > 0.06:  # 6% profit
            return 0.03   # Lock in 3% minimum
            
        # Big profit trailing
        if current_profit > 0.10:  # 10% profit
            return 0.05   # Lock in 5% minimum

        return max(self.stoploss, dynamic_sl)

def custom_stoploss(self, pair: str, trade: Trade, current_time: datetime,
