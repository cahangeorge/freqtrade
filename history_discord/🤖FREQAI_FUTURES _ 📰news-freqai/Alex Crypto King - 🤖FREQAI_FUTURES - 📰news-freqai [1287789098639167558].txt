def calculate_eo(A, omega, t, omega_n, n):
    # First term: A * cos(omega * t)
    term_1 = A * np.cos(omega * t)
    
    # Second term: Summation of 2A * cos(omega * t) * cos(k * omega_n) for k in the range 1 to (n-1)/2
    summation_term = 0
    for k in range(1, (n - 1) // 2 + 1):
        summation_term += np.cos(k * omega_n)
    
    # Multiply the summation by 2A * cos(omega * t)
    term_2 = 2 * A * np.cos(omega * t) * summation_term
    
    # Total expression
    eo = (1 / n) * (term_1 + term_2)
    
    return eo

def calculate_f(omega_n, n):
    # Initialize f to 0
    f = 0
