[01/12/2026 17:03] vasko123.
def leverage(self, pair: str, current_time: datetime, current_rate: float,
                 proposed_leverage: float, max_leverage: float, entry_tag: Optional[str],
                 side: str, **kwargs) -> float:

        # Base fallback from strategy constant
        lev = float(self.leverage_value)  # ✅ Explicit conversion to Python float

        # Tag-based scaling
        if entry_tag:
            if "_Super" in entry_tag:
                lev *= 1.4   # +40%
            elif "_HQ" in entry_tag:
                lev *= 1.0   # unchanged
            elif "_Early" in entry_tag:
                lev *= 0.7   # -30%
            elif "_Weak" in entry_tag:
                lev = 0.8 * float(self.leverage_value)  # ✅ Force float      smaller exposure
        # Count-based fine-tuning
        try:
            dataframe, _ = self.dp.get_analyzed_dataframe(pair, self.timeframe)
            if dataframe is not None and not dataframe.empty:
                last_row = dataframe.iloc[-1]
                div_strength = max(
                    float(last_row.get('total_bullish_divergences_count', 0)),  # ✅ Convert
                    float(last_row.get('total_bearish_divergences_count', 0))   # ✅ Convert
                )
                # small boost (up to +1x)
                lev += min(div_strength * 0.2, 1.0)
        except Exception:
            pass

        # Clamp to max allowed leverage
        return float(min(max_leverage, round(lev, 2)))  # ✅ Final conversion
